cmake_minimum_required(VERSION 3.14)
project(p101_json_tests C)
enable_testing()
set(CMAKE_C_STANDARD 17)
add_library(error_under_test STATIC ../../lib_error/src/error.c)
target_include_directories(error_under_test PUBLIC ../../lib_error/include)
add_library(json_under_test STATIC ../src/json.c)
target_compile_definitions(json_under_test PRIVATE P101_JSON_TESTING=1)
target_include_directories(json_under_test PUBLIC ../include ../../lib_error/include)
target_link_libraries(json_under_test PUBLIC error_under_test)
add_executable(test_json test_json.c)
target_link_libraries(test_json PRIVATE json_under_test)
add_test(NAME test_json COMMAND test_json)
